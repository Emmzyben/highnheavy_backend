CREATE TABLE `bookings` (
  `id` char(36) NOT NULL DEFAULT uuid(),
  `shipper_id` char(36) NOT NULL,
  `carrier_id` char(36) DEFAULT NULL,
  `escort_id` char(36) DEFAULT NULL,
  `assigned_driver_id` char(36) DEFAULT NULL,
  `status` varchar(50) NOT NULL DEFAULT 'pending_quote' CHECK (`status` in ('pending_quote','quoted','awaiting_payment','booked','in_transit','delivered','cancelled','completed')),
  `delivery_photos` longtext CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL CHECK (json_valid(`delivery_photos`)),
  `delivery_signature` text DEFAULT NULL,
  `receiver_name` varchar(255) DEFAULT NULL,
  `pickup_address` text NOT NULL,
  `pickup_city` varchar(100) NOT NULL,
  `pickup_state` varchar(50) NOT NULL,
  `delivery_address` text NOT NULL,
  `delivery_city` varchar(100) NOT NULL,
  `delivery_state` varchar(50) NOT NULL,
  `shipment_date` date NOT NULL,
  `flexible_dates` tinyint(1) DEFAULT 0,
  `requires_escort` tinyint(1) DEFAULT 0,
  `special_instructions` text DEFAULT NULL,
  `cargo_type` varchar(100) NOT NULL,
  `cargo_description` text NOT NULL,
  `dimensions_length_ft` decimal(10,2) NOT NULL,
  `dimensions_width_ft` decimal(10,2) NOT NULL,
  `dimensions_height_ft` decimal(10,2) NOT NULL,
  `weight_lbs` decimal(10,2) NOT NULL,
  `weight_unit` varchar(10) DEFAULT 'lbs',
  `delivery_date_est` date DEFAULT NULL,
  `agreed_price` decimal(12,2) DEFAULT NULL,
  `currency` varchar(3) DEFAULT 'USD',
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  `pickup_latitude` decimal(10,8) DEFAULT NULL COMMENT 'Pickup location latitude',
  `pickup_longitude` decimal(11,8) DEFAULT NULL COMMENT 'Pickup location longitude',
  `delivery_latitude` decimal(10,8) DEFAULT NULL COMMENT 'Delivery location latitude',
  `delivery_longitude` decimal(11,8) DEFAULT NULL COMMENT 'Delivery location longitude',
  `markup_type` enum('percentage','lump_sum') DEFAULT 'percentage',
  `markup_value` decimal(12,2) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `shipper_id` (`shipper_id`),
  KEY `carrier_id` (`carrier_id`),
  KEY `escort_id` (`escort_id`),
  KEY `fk_booking_driver` (`assigned_driver_id`),
  CONSTRAINT `bookings_ibfk_1` FOREIGN KEY (`shipper_id`) REFERENCES `users` (`id`),
  CONSTRAINT `bookings_ibfk_2` FOREIGN KEY (`carrier_id`) REFERENCES `users` (`id`),
  CONSTRAINT `bookings_ibfk_3` FOREIGN KEY (`escort_id`) REFERENCES `users` (`id`),
  CONSTRAINT `fk_booking_driver` FOREIGN KEY (`assigned_driver_id`) REFERENCES `drivers` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci
